// worker picks ports from the jobs channel and tests connectivity.
func scanWorker(_ int, ports <-chan int, results chan<- string, targetHost string) {
        for port := range ports {
                address := fmt.Sprintf("%s:%d", targetHost, port)
                conn, err := net.DialTimeout("tcp", address, time.Second)
                if err != nil {
                        results <- fmt.Sprintf("[-] Port %d: CLOSED", port)
                        continue
                }
                conn.Close()
                results <- fmt.Sprintf("[+] Port %d: OPEN", port)
        }
}

// RunPortScan scans an authorized target over the requested port range.
func RunAuthorizedPortScan(targetHost string, startPort, endPort int, workerCount int) {
        ports := make(chan int, 100)
        results := make(chan string, 100)

        var wg sync.WaitGroup

        for workerID := 1; workerID <= workerCount; workerID++ {
                wg.Add(1)
                go func(id int) {
                        defer wg.Done()
                        scanWorker(id, ports, results, targetHost)
                }(workerID)
        }

        go func() {
                for port := startPort; port <= endPort; port++ {
                        ports <- port
                }
                close(ports)
        }()

        go func() {
                wg.Wait()
                close(results)
        }()

        fmt.Printf("[*] Starting concurrent scan on host: %s (Ports %d-%d)\n", targetHost, startPort, endPort)
        for result := range results {
                fmt.Println(result)
        }
        fmt.Println("[*] Port scan completed.")
}

const (
        defaultCarveBufferSize = 1 << 20
        maxCarveBufferSize     = 16 << 20
        defaultMaxELFSize      = 15 << 20
)

type CarveConfig struct {
        ImageFilePath string
        OutputFolder  string
        BufferSize    int64
        MaxELFSize    int64
}

type MagicNumber struct {
        Extension string
        Header    []byte
        Footer    []byte
}

type RecoveryEvent struct {
        Timestamp     string `json:"timestamp"`
        Source        string `json:"source"`
        EventType     string `json:"event_type"`
        FileType      string `json:"file_type"`
        OffsetStart   int64  `json:"offset_start"`
        FileSizeBytes int64  `json:"file_size_bytes"`
        SHA256Hash    string `json:"sha256_hash"`
}

var knownSignatures = []MagicNumber{
        {Extension: "jpg", Header: []byte{0xFF, 0xD8, 0xFF}, Footer: []byte{0xFF, 0xD9}},
        {Extension: "png", Header: []byte{0x89, 0x50, 0x4E, 0x47, 0x0D, 0x0A, 0x1A, 0x0A}, Footer: []byte{0x49, 0x45, 0x4E, 0x44, 0xAE, 0x42, 0x60, 0x82}},
        {Extension: "pdf", Header: []byte{0x25, 0x50, 0x44, 0x46}, Footer: []byte{0x25, 0x25, 0x45, 0x4F, 0x46}},
        {Extension: "zip", Header: []byte{0x50, 0x4B, 0x03, 0x04}, Footer: []byte{0x50, 0x4B, 0x05, 0x06}},
        {Extension: "elf", Header: []byte{0x7F, 0x45, 0x4C, 0x46}},
}

// RunCarver scans a regular disk-image file without modifying it and writes
// recovered files to a separate output directory.
func RunCarver(config CarveConfig) ([]RecoveryEvent, error) {
        if config.ImageFilePath == "" || config.OutputFolder == "" {
                return nil, fmt.Errorf("image file and output folder are required")
        }
        if config.BufferSize == 0 {
                config.BufferSize = defaultCarveBufferSize
        }
        if config.BufferSize < 1 || config.BufferSize > maxCarveBufferSize {
                return nil, fmt.Errorf("buffer size must be between 1 byte and %d bytes", maxCarveBufferSize)
        }
        if config.MaxELFSize == 0 {
                config.MaxELFSize = defaultMaxELFSize
        }
        if config.MaxELFSize < 1 {
                return nil, fmt.Errorf("maximum ELF size must be positive")
        }

        imagePath, err := filepath.Abs(config.ImageFilePath)
        if err != nil {
                return nil, fmt.Errorf("resolve image path: %w", err)
        }
        image, err := os.Open(imagePath)
        if err != nil {
                return nil, fmt.Errorf("open image file: %w", err)
        }
        defer image.Close()

        imageInfo, err := image.Stat()
        if err != nil {
                return nil, fmt.Errorf("inspect image file: %w", err)
        }
        if !imageInfo.Mode().IsRegular() {
                return nil, fmt.Errorf("image path must refer to a regular file")
        }
        imagePath, err = filepath.EvalSymlinks(imagePath)
        if err != nil {
                return nil, fmt.Errorf("resolve image file: %w", err)
        }

        outputPath, err := filepath.Abs(config.OutputFolder)
        if err != nil {
                return nil, fmt.Errorf("resolve output folder: %w", err)
        }
        if err := os.MkdirAll(outputPath, 0700); err != nil {
                return nil, fmt.Errorf("create output folder: %w", err)
        }
        outputPath, err = filepath.EvalSymlinks(outputPath)
        if err != nil {
                return nil, fmt.Errorf("resolve output folder: %w", err)
        }
        if filepath.Clean(outputPath) == filepath.Dir(imagePath) {
                return nil, fmt.Errorf("output folder must be separate from the image folder")
        }

        return carveImage(image, imageInfo.Size(), outputPath, config)
}

func carveImage(image *os.File, imageSize int64, outputFolder string, config CarveConfig) ([]RecoveryEvent, error) {
        var events []RecoveryEvent
        for _, signature := range knownSignatures {
                if signature.Footer != nil {
                        carved, err := carveFooterTerminatedFiles(image, imageSize, outputFolder, signature, config.BufferSize)
                        if err != nil {
                                return events, err
                        }
                        events = append(events, carved...)
                        continue
                }

                patterns := [][]byte{signature.Header}
                searchFrom := int64(0)
                for searchFrom < imageSize {
                        start, _, found, err := findNextPattern(image, patterns, searchFrom, imageSize, config.BufferSize)
                        if err != nil {
                                return events, fmt.Errorf("scan for %s header: %w", signature.Extension, err)
                        }
                        if !found {
                                break
                        }

                        end := start + config.MaxELFSize
                        if end > imageSize {
                                end = imageSize
                        }

                        event, err := writeCarvedFile(image, outputFolder, signature.Extension, start, end)
                        if err != nil {
                                return events, err
                        }
                        events = append(events, event)
                        searchFrom = end
                }
        }
        return events, nil
}

func carveFooterTerminatedFiles(image *os.File, imageSize int64, outputFolder string, signature MagicNumber, bufferSize int64) ([]RecoveryEvent, error) {
        var events []RecoveryEvent
        patterns := [][]byte{signature.Header, signature.Footer}
        var pendingStart int64 = -1
        searchFrom := int64(0)

        for searchFrom < imageSize {
                offset, patternIndex, found, err := findNextPattern(image, patterns, searchFrom, imageSize, bufferSize)
                if err != nil {
                        return events, fmt.Errorf("scan %s signatures: %w", signature.Extension, err)
                }
                if !found {
                        break
                }
                if patternIndex == 0 {
                        if pendingStart < 0 {
                                pendingStart = offset
                        }
                } else if pendingStart >= 0 {
                        end := offset + int64(len(signature.Footer))
                        event, err := writeCarvedFile(image, outputFolder, signature.Extension, pendingStart, end)
                        if err != nil {
                                return events, err
                        }
                        events = append(events, event)
                        pendingStart = -1
                }
                searchFrom = offset + int64(len(patterns[patternIndex]))
        }
        return events, nil
}

func findNextPattern(image *os.File, patterns [][]byte, start, imageSize, bufferSize int64) (int64, int, bool, error) {
        maxPatternLength := 0
        for _, pattern := range patterns {
                if len(pattern) == 0 {
                        return 0, 0, false, fmt.Errorf("patterns must not be empty")
                }
                if len(pattern) > maxPatternLength {
                        maxPatternLength = len(pattern)
                }
        }

        buffer := make([]byte, int(bufferSize))
        var carry []byte
        offset := start
        for offset < imageSize {
                readSize := int64(len(buffer))
                if remaining := imageSize - offset; remaining < readSize {
                        readSize = remaining
                }
                n, readErr := image.ReadAt(buffer[:int(readSize)], offset)
                if n > 0 {
                        chunk := buffer[:n]
                        combined := make([]byte, len(carry)+len(chunk))
                        copy(combined, carry)
                        copy(combined[len(carry):], chunk)
                        bestIndex, bestPattern := -1, -1
                        for i, pattern := range patterns {
                                if index := bytes.Index(combined, pattern); index >= 0 && (bestIndex < 0 || index < bestIndex) {
                                        bestIndex, bestPattern = index, i
                                }
                        }
                        if bestIndex >= 0 {
                                return offset - int64(len(carry)) + int64(bestIndex), bestPattern, true, nil
                        }

                        carrySize := maxPatternLength - 1
                        if carrySize > len(combined) {
                                carrySize = len(combined)
                        }
                        carry = append(carry[:0], combined[len(combined)-carrySize:]...)
                        offset += int64(n)
                }
                if readErr != nil {
                        if readErr == io.EOF {
                                return 0, 0, false, nil
                        }
                        return 0, 0, false, readErr
                }
                if n == 0 {
                        return 0, 0, false, nil
                }
        }
        return 0, 0, false, nil
}

func writeCarvedFile(image *os.File, outputFolder, extension string, start, end int64) (RecoveryEvent, error) {
        size := end - start
        if size <= 0 {
                return RecoveryEvent{}, fmt.Errorf("invalid %s carve range at offset %d", extension, start)
        }
        outputPath := filepath.Join(outputFolder, fmt.Sprintf("recovered_%s_%012d.%s", extension, start, extension))
        output, err := os.OpenFile(outputPath, os.O_WRONLY|os.O_CREATE|os.O_EXCL, 0600)
        if err != nil {
                return RecoveryEvent{}, fmt.Errorf("create recovered file at offset %d: %w", start, err)
        }

        hash := sha256.New()
        reader := io.NewSectionReader(image, start, size)
        _, copyErr := io.CopyN(io.MultiWriter(output, hash), reader, size)
        closeErr := output.Close()
        if copyErr != nil {
                if removeErr := os.Remove(outputPath); removeErr != nil {
                        return RecoveryEvent{}, fmt.Errorf("write recovered file at offset %d: %w (also failed to remove partial file: %v)", start, copyErr, removeErr)
                }
                return RecoveryEvent{}, fmt.Errorf("write recovered file at offset %d: %w", start, copyErr)
        }
        if closeErr != nil {
                if removeErr := os.Remove(outputPath); removeErr != nil {
                        return RecoveryEvent{}, fmt.Errorf("close recovered file at offset %d: %w (also failed to remove partial file: %v)", start, closeErr, removeErr)
                }
                return RecoveryEvent{}, fmt.Errorf("close recovered file at offset %d: %w", start, closeErr)
        }

        return RecoveryEvent{
                Timestamp:     time.Now().UTC().Format(time.RFC3339),
                Source:        "Ovim-Gallery-Carver",
                EventType:     "file_recovered",
                FileType:      extension,
                OffsetStart:   start,
                FileSizeBytes: size,
                SHA256Hash:    hex.EncodeToString(hash.Sum(nil)),
        }, nil
}

// ValidateJPEG checks the JPEG start and end markers. The 15 MB size guidance
// is not enforced as a hard limit.
func ValidateJPEG(data []byte) error {
        if len(data) < 4 {
                return fmt.Errorf("JPEG data is too short")
        }
        if data[0] != 0xFF || data[1] != 0xD8 || data[2] != 0xFF {
                return fmt.Errorf("invalid JPEG header")
        }
        if data[len(data)-2] != 0xFF || data[len(data)-1] != 0xD9 {
                return fmt.Errorf("invalid JPEG footer")
        }
        return nil
}

// sendLog transmits a JSON payload to the Splunk HEC endpoint with retry logic and token auth.
func sendLog(target, token string, client *http.Client, payload []byte) (*http.Response, error) {
        var lastErr error

        for attempt := 1; attempt <= 3; attempt++ {
                req, err := http.NewRequest(http.MethodPost, target, bytes.NewBuffer(payload))
                if err != nil {
                        return nil, err
                }
                // Set required headers for Splunk HEC.
                req.Header.Set("Content-Type", "application/json")
                if token != "" {
                        req.Header.Set("Authorization", "Splunk "+token)
                }

                resp, err := client.Do(req)
                if err == nil && resp.StatusCode == http.StatusOK {
                        return resp, nil
                }

                if resp != nil {
                        lastErr = fmt.Errorf("unexpected response status: %s", resp.Status)
                        resp.Body.Close()
                } else {
                        lastErr = err
                }

                fmt.Printf("[!] Attempt %d failed. Retrying in %d seconds...\n", attempt, attempt*2)
                time.Sleep(time.Duration(attempt*2) * time.Second)
        }

        return nil, lastErr
}

// SendLogToSplunkHEC sends a security event to the Splunk HEC event endpoint.
func SendLogToSplunkHEC(splunkURL, hecToken, eventType, status string) (*http.Response, error) {
        now := time.Now().UTC()
        payload, err := json.Marshal(hecEvent{
                Time:       now.Unix(),
                Host:       "local-scanner-node",
                Sourcetype: "ovimscannerjson",
                Event: SecurityEvent{
                        Timestamp: now.Format(time.RFC3339),
                        EventType: eventType,
                        Status:    status,
                },
        })
        if err != nil {
                return nil, fmt.Errorf("failed to marshal HEC payload: %w", err)
        }

        client := &http.Client{Timeout: 10 * time.Second}
        return sendLog(splunkURL, hecToken, client, payload)
}

        trackingServerFlag := flag.Bool("tracking-server", false, "Start a loopback-only tracking endpoint at 127.0.0.1:8080/track")
        carveImageFlag := flag.String("carve-image", "", "Scan a regular image file for recoverable files")
        carveOutputFlag := flag.String("carve-output", "", "Output directory for recovered files (must differ from image directory)")
        carveELFMaxBytesFlag := flag.Int64("carve-elf-max-bytes", defaultMaxELFSize, "Maximum bytes to carve per ELF file")
        flag.Parse()

        carveSelected := *carveImageFlag != "" || *carveOutputFlag != ""
        if carveSelected && (*carveImageFlag == "" || *carveOutputFlag == "") {
                log.Fatal("[-] -carve-image and -carve-output must be provided together")
        }
        selectedIntegrations := 0
        for _, selected := range []bool{*xbowFindingsFlag, *nodeZeroPathsFlag, *nodeZeroToSplunkFlag, *trackingServerFlag, carveSelected} {
                if selected {
                        selectedIntegrations++
                }
        }
        if selectedIntegrations > 1 {
                log.Fatal("[-] Choose only one security findings integration per run")
        }

        if *xbowFindingsFlag {
                client := NewXBOWClient(os.Getenv("XBOW_API_KEY"))
                findings, err := client.FetchAndLogFindings(*xbowAssetIDFlag)
                if err != nil {
                        log.Fatalf("[-] Failed to fetch XBOW findings: %v", err)
                }
                log.Printf("[+] Logged %d XBOW findings for asset %s", len(findings), *xbowAssetIDFlag)
                return
        }
        if *nodeZeroPathsFlag {
                log.Println("Initializing NodeZero security telemetry collector...")
                client := NewNodeZeroClient()
                paths, err := client.FetchAndLogAttackPaths()
                if err != nil {
                        log.Fatalf("Failed to fetch and log NodeZero attack paths: %v", err)
                }
                log.Printf("Successfully processed and logged %d attack paths.", len(paths))
                return
        }
        if *nodeZeroToSplunkFlag {
                log.Println("Fetching NodeZero attack-path metadata...")
                client := NewNodeZeroClient()
                paths, err := client.FetchDetailedAttackPaths()
                if err != nil {
                        log.Fatalf("[-] Failed to fetch NodeZero attack paths: %v", err)

                if err := ShipAttackPathsToSplunk(paths); err != nil {
                        log.Fatalf("[-] Failed to ship NodeZero attack paths to Splunk: %v", err)
                }
                log.Printf("[+] Shipped %d NodeZero attack paths to Splunk", len(paths))
                return
        }
        if *trackingServerFlag {
                log.Println("Starting tracking endpoint at http://127.0.0.1:8080/track")
                if err := http.ListenAndServe("127.0.0.1:8080", newTrackingHandler()); err != nil {
                        log.Fatalf("[-] Tracking server failed: %v", err)
                }
                return
        }
        if carveSelected {
                events, carveErr := RunCarver(CarveConfig{
                        ImageFilePath: *carveImageFlag,
                        OutputFolder:  *carveOutputFlag,
                        MaxELFSize:    *carveELFMaxBytesFlag,

                                log.Printf("[-] Failed to ship partial recovery metadata: %v", shipErr)

                        log.Fatalf("[-] File carving failed: %v", carveErr)
                }

                return
        }

        fmt.Printf("[*] Initializing log transmission to target: %s\n", *targetFlag)

        if err != nil {
                log.Fatalf("[-] Failed to send log after all attempts: %v", err)
        }
        defer resp.Body.Close()

        fmt.Printf("[+] Successfully sent log to Splunk. Response Status: %s\n", resp.Status)
}

func validateNodeZeroURL(rawURL string) error {
        parsed, err := url.Parse(rawURL)
        if err != nil {
                return fmt.Errorf("invalid NodeZero URL: %w", err)
        }
        if parsed.Hostname() == "" || parsed.User != nil || parsed.RawQuery != "" ||
                parsed.ForceQuery || parsed.Fragment != "" {
                return fmt.Errorf("invalid NodeZero URL")
        }
        if parsed.Scheme == "https" {
                return nil
        }
        if parsed.Scheme == "http" {
                host := parsed.Hostname()
                if strings.EqualFold(host, "localhost") {
                        return nil
                }
                if ip := net.ParseIP(host); ip != nil && ip.Equal(net.IPv4(127, 0, 0, 1)) {
                        return nil
                }
        }
        return fmt.Errorf("NodeZero URLs must use HTTPS (HTTP is allowed only for loopback tests)")
}

func (c *NodeZeroClient) postJSON(endpoint string, payload any, bearerToken string) ([]byte, error) {
        if err := validateNodeZeroURL(endpoint); err != nil {
                return nil, err
        }
        body, err := json.Marshal(payload)
        if err != nil {
                return nil, fmt.Errorf("encode NodeZero request: %w", err)
        }
        req, err := http.NewRequest(http.MethodPost, endpoint, bytes.NewReader(body))
        if err != nil {
                return nil, fmt.Errorf("create NodeZero request: %w", err)
        }
        req.Header.Set("Content-Type", "application/json")
        req.Header.Set("Accept", "application/json")
        if bearerToken != "" {
                req.Header.Set("Authorization", "Bearer "+bearerToken)
        }

        resp, err := c.HTTPClient.Do(req)
        if err != nil {
                return nil, fmt.Errorf("send NodeZero request: %w", err)
        }
        defer resp.Body.Close()
        if resp.StatusCode < http.StatusOK || resp.StatusCode >= http.StatusMultipleChoices {
                return nil, fmt.Errorf("NodeZero request returned HTTP status %s", resp.Status)
        }
        responseBody, err := io.ReadAll(io.LimitReader(resp.Body, maxNodeZeroResponseBytes+1))
        if err != nil {
                return nil, fmt.Errorf("read NodeZero response: %w", err)
        }
        if len(responseBody) > maxNodeZeroResponseBytes {
                return nil, fmt.Errorf("NodeZero response exceeds %d bytes", maxNodeZeroResponseBytes)
        }
        return responseBody, nil
}

func (c *NodeZeroClient) Authenticate() (string, error) {
        if strings.TrimSpace(c.APIKey) == "" {
                return "", fmt.Errorf("H3_API_KEY is required")
        }
        if c.HTTPClient == nil {
                return "", fmt.Errorf("NodeZero HTTP client is not configured")
        }

        responseBody, err := c.postJSON(c.AuthURL, AuthRequest{Key: c.APIKey}, "")
        if err != nil {
                return "", fmt.Errorf("authenticate with NodeZero: %w", err)
        }
        var authResponse AuthResponse
        if err := json.Unmarshal(responseBody, &authResponse); err != nil {
                return "", fmt.Errorf("decode NodeZero authentication response: %w", err)
        }
        if strings.TrimSpace(authResponse.Token) == "" {
                return "", fmt.Errorf("NodeZero authentication response did not contain a token")
        }
        return authResponse.Token, nil
}

// worker picks ports from the jobs channel and tests connectivity.
func scanPortWorker(_ int, ports <-chan int, results chan<- string, targetHost string) {
        for port := range ports {
                address := fmt.Sprintf("%s:%d", targetHost, port)
                conn, err := net.DialTimeout("tcp", address, time.Second)
                if err != nil {
                        results <- fmt.Sprintf("[-] Port %d: CLOSED", port)
                        continue
                }
                conn.Close()
                results <- fmt.Sprintf("[+] Port %d: OPEN", port)
        }
}

// RunPortScan scans an authorized target over the requested port range.
func RunPortScan(targetHost string, startPort, endPort int, workerCount int) {
        if startPort > endPort {
                startPort, endPort = endPort, startPort
        }
        if workerCount <= 0 {
                workerCount = 1
        }

        ports := make(chan int, 100)
        results := make(chan string, 100)

        var wg sync.WaitGroup

        for workerID := 1; workerID <= workerCount; workerID++ {
                wg.Add(1)
                go func(id int) {
                        defer wg.Done()
                        scanPortWorker(id, ports, results, targetHost)
                }(workerID)
        }

        go func() {
                defer close(ports)
                for port := startPort; port <= endPort; port++ {
                        ports <- port
                }
        }()

        go func() {
                wg.Wait()
                close(results)
        }()

        fmt.Printf("[*] Starting concurrent scan on host: %s (Ports %d-%d)\n", targetHost, startPort, endPort)
        for result := range results {
                fmt.Println(result)
        }
        fmt.Println("[*] Port scan completed.")
}

type SplunkHECPayload struct {
        Time       int64           `json:"time"`
        Host       string          `json:"host"`
        Source     string          `json:"source,omitempty"`
        SourceType string          `json:"sourcetype"`

// ShipAttackPathsToSplunk sends detailed NodeZero attack-path metadata to Splunk HEC.
func ShipAttackPathsToSplunk(paths []DetailedAttackPath) error {
        splunkURL := os.Getenv("SPLUNK_HEC_URL")
        splunkToken := os.Getenv("SPLUNK_HEC_TOKEN")
        if strings.TrimSpace(splunkURL) == "" || strings.TrimSpace(splunkToken) == "" {
                return fmt.Errorf("SPLUNK_HEC_URL and SPLUNK_HEC_TOKEN environment variables must be set")
        }

        endpoint, err := splunkHECEndpoint(splunkURL)
        if err != nil {
                return err
        }
        client := &http.Client{Timeout: 10 * time.Second}

        for _, path := range paths {
                event, err := json.Marshal(path)
                if err != nil {
                        return fmt.Errorf("encode NodeZero attack path %q: %w", path.ID, err)
                }
                body, err := json.Marshal(SplunkHECPayload{
                        Time:       time.Now().Unix(),
                        Host:       "nodezero-scanner-01",
                        Source:     "nodezero_ai_offensive",
                        SourceType: "_json",
                        Event:      event,
                })
                if err != nil {
                        return fmt.Errorf("encode Splunk HEC payload for NodeZero attack path %q: %w", path.ID, err)
                }

                req, err := http.NewRequest(http.MethodPost, endpoint, bytes.NewReader(body))
                if err != nil {
                        return fmt.Errorf("create Splunk HEC request for NodeZero attack path %q: %w", path.ID, err)
                }
                req.Header.Set("Authorization", "Splunk "+splunkToken)
                req.Header.Set("Content-Type", "application/json")

                resp, err := client.Do(req)
                if err != nil {
                        return fmt.Errorf("send NodeZero attack path %q to Splunk HEC: %w", path.ID, err)
                }
                _, readErr := io.Copy(io.Discard, resp.Body)
                closeErr := resp.Body.Close()
                if readErr != nil {
                        return fmt.Errorf("read Splunk HEC response for NodeZero attack path %q: %w", path.ID, readErr)
                }
                if closeErr != nil {
                        return fmt.Errorf("close Splunk HEC response for NodeZero attack path %q: %w", path.ID, closeErr)
                }
                if resp.StatusCode < http.StatusOK || resp.StatusCode >= http.StatusMultipleChoices {
                        return fmt.Errorf("Splunk HEC rejected NodeZero attack path %q: unexpected HTTP status %s", path.ID, resp.Status)
                }
        }
        return nil
}

func splunkHECEndpoint(rawURL string) (string, error) {
        parsed, err := url.Parse(rawURL)
        if err != nil || parsed.Host == "" ||
                (parsed.Scheme != "https" && parsed.Scheme != "http") ||
                parsed.User != nil || parsed.RawQuery != "" || parsed.Fragment != "" {
                return "", fmt.Errorf("invalid SPLUNK_HEC_URL")
        }
        if !strings.HasSuffix(strings.TrimRight(parsed.Path, "/"), "/services/collector") {
                parsed.Path = strings.TrimRight(parsed.Path, "/") + "/services/collector"
        }
        return parsed.String(), nil
}

// ShipLogFileToSplunk sends each JSON object in a JSONL file to Splunk HEC.
func ShipLogFileToSplunk(logPath, hecURL, hecToken string) error {
        file, err := os.Open(logPath)
        if err != nil {
                return fmt.Errorf("open security event log: %w", err)
        }
        defer file.Close()

        scanner := bufio.NewScanner(file)
        scanner.Buffer(make([]byte, 64*1024), maxSplunkEventLineBytes)
        client := &http.Client{Timeout: 10 * time.Second}

        for lineNumber := 1; scanner.Scan(); lineNumber++ {
                line := bytes.TrimSpace(scanner.Bytes())
                if len(line) == 0 {
                        continue
                }

                var event map[string]json.RawMessage
                if err := json.Unmarshal(line, &event); err != nil {
                        return fmt.Errorf("decode security event log line %d: %w", lineNumber, err)
                }
                if event == nil {
                        return fmt.Errorf("decode security event log line %d: expected a JSON object", lineNumber)
                }

                payload := SplunkHECPayload{
                        Time:       time.Now().Unix(),
                        Host:       "local-scanner-node",
                        SourceType: "xbow:findings:json",
                        Event:      append(json.RawMessage(nil), line...),
                }
                body, err := json.Marshal(payload)
                if err != nil {
                        return fmt.Errorf("encode Splunk HEC payload for line %d: %w", lineNumber, err)
                }

                req, err := http.NewRequest(http.MethodPost, hecURL, bytes.NewReader(body))
                if err != nil {
                        return fmt.Errorf("create Splunk HEC request for line %d: %w", lineNumber, err)
                }
                req.Header.Set("Authorization", "Splunk "+hecToken)
                req.Header.Set("Content-Type", "application/json")

                resp, err := client.Do(req)
                if err != nil {
                        return fmt.Errorf("ship security event on line %d: %w", lineNumber, err)
                }
                resp.Body.Close()
                if resp.StatusCode < http.StatusOK || resp.StatusCode >= http.StatusMultipleChoices {
                        return fmt.Errorf("ship security event on line %d: unexpected HTTP status %s", lineNumber, resp.Status)
                }
        }
        if err := scanner.Err(); err != nil {
                return fmt.Errorf("read security event log: %w", err)
        }
        return nil
}

func GenerateTrackingToken(length int) (string, error) {
        if length <= 0 {
                return "", fmt.Errorf("tracking token length must be positive")
        }
        bytes := make([]byte, length)
        if _, err := rand.Read(bytes); err != nil {
                return "", err
        }
        return hex.EncodeToString(bytes), nil
}

type trackingLinkStore struct {
        mu    sync.Mutex
        links map[string]TrackingLink
}

type TrackingLink struct {
        ID         string    `json:"id"`
        TargetURL  string    `json:"target_url"`
        CreatedAt  time.Time `json:"created_at"`
        ClickCount int       `json:"click_count"`
}

var defaultTrackingLinkStore = &trackingLinkStore{links: make(map[string]TrackingLink)}

func CreateTrackingLink(targetURL string) (string, error) {
        parsedURL, err := url.Parse(targetURL)
        if err != nil || !parsedURL.IsAbs() || parsedURL.Host == "" ||
                (strings.ToLower(parsedURL.Scheme) != "http" && strings.ToLower(parsedURL.Scheme) != "https") || parsedURL.User != nil {
                return "", fmt.Errorf("tracking target must be an absolute HTTP or HTTPS URL")
        }
        parsedURL.Scheme = strings.ToLower(parsedURL.Scheme)

        for {
                token, err := GenerateTrackingToken(32)
                if err != nil {
                        return "", fmt.Errorf("generate tracking token: %w", err)
                }

                defaultTrackingLinkStore.mu.Lock()
                if _, exists := defaultTrackingLinkStore.links[token]; !exists {
                        defaultTrackingLinkStore.links[token] = TrackingLink{
                                ID:        token,
                                TargetURL: parsedURL.String(),
                                CreatedAt: time.Now().UTC(),
                        }
                        defaultTrackingLinkStore.mu.Unlock()
                        return token, nil
                }
                defaultTrackingLinkStore.mu.Unlock()
        }
}

func HandleTrackClick(w http.ResponseWriter, r *http.Request) {
        handleTrackClick(defaultTrackingLinkStore, w, r)
}

func handleTrackClick(store *trackingLinkStore, w http.ResponseWriter, r *http.Request) {
        token := r.URL.Query().Get("token")
        if token == "" {
                http.Error(w, "Missing tracking token", http.StatusBadRequest)
                return
        }

        store.mu.Lock()
        link, exists := store.links[token]
        if exists {
                link.ClickCount++
                store.links[token] = link
        }
        store.mu.Unlock()

        if !exists {
                http.NotFound(w, r)
                return
        }

        clientIP := "unknown"
        if host, _, err := net.SplitHostPort(r.RemoteAddr); err == nil && net.ParseIP(host) != nil {
                clientIP = host
        } else if net.ParseIP(r.RemoteAddr) != nil {
                clientIP = r.RemoteAddr
        }
        log.Printf("Tracking click token=%s ip=%s user_agent=%q timestamp=%s",
                token, clientIP, r.UserAgent(), time.Now().UTC().Format(time.RFC3339Nano))
        http.Redirect(w, r, link.TargetURL, http.StatusFound)
}

func handleTrackingRequest(w http.ResponseWriter, r *http.Request) {
        clientIP := "unknown"
        if host, _, err := net.SplitHostPort(r.RemoteAddr); err == nil && net.ParseIP(host) != nil {
                clientIP = host
        } else if net.ParseIP(r.RemoteAddr) != nil {
                clientIP = r.RemoteAddr
        }

        log.Printf("Received tracking request from IP: %s", clientIP)
        w.Header().Set("Content-Type", "text/plain; charset=utf-8")
        w.WriteHeader(http.StatusOK)
        if _, err := fmt.Fprint(w, "Access Logged"); err != nil {
                log.Printf("Failed to write tracking response: %v", err)
        }
}

func newTrackingHandler() http.Handler {
        mux := http.NewServeMux()
        mux.HandleFunc("/track", handleTrackingRequest)
        mux.HandleFunc("/click", HandleTrackClick)
        return mux
}

// FetchAndLogFindings retrieves findings for one asset and appends them as JSONL.
func (x *XBOWClient) FetchAndLogFindings(assetID string) ([]Finding, error) {
        if strings.TrimSpace(x.APIKey) == "" {
                return nil, fmt.Errorf("XBOW API key is required")
        }
        if assetID == "" || assetID == "." || assetID == ".." {
                return nil, fmt.Errorf("invalid XBOW asset ID")
        }
        for _, r := range assetID {
                if !((r >= 'a' && r <= 'z') || (r >= 'A' && r <= 'Z') ||
                        (r >= '0' && r <= '9') || strings.ContainsRune("._~-", r)) {
                        return nil, fmt.Errorf("invalid XBOW asset ID")
                }
        }
        if x.Client == nil {
                return nil, fmt.Errorf("XBOW HTTP client is not configured")
        }
        if x.BaseURL == "" {
                return nil, fmt.Errorf("XBOW API base URL is not configured")
        }

        endpoint := strings.TrimRight(x.BaseURL, "/") + "/assets/" + url.PathEscape(assetID) + "/findings"
        req, err := http.NewRequest(http.MethodGet, endpoint, nil)
        if err != nil {
                return nil, fmt.Errorf("create XBOW findings request: %w", err)
        }
        req.Header.Set("Authorization", "Bearer "+x.APIKey)
        req.Header.Set("X-XBOW-API-Version", xbowAPIVersion)
        req.Header.Set("Accept", "application/json")

        resp, err := x.Client.Do(req)
        if err != nil {
                return nil, fmt.Errorf("fetch XBOW findings: %w", err)
        }
        defer resp.Body.Close()

        if resp.StatusCode < http.StatusOK || resp.StatusCode >= http.StatusMultipleChoices {
                return nil, fmt.Errorf("fetch XBOW findings: unexpected HTTP status %s", resp.Status)
        }

        responseBody, err := io.ReadAll(io.LimitReader(resp.Body, maxXBOWResponseBytes+1))
        if err != nil {
                return nil, fmt.Errorf("read XBOW findings response: %w", err)
        }
        if len(responseBody) > maxXBOWResponseBytes {
                return nil, fmt.Errorf("XBOW findings response exceeds %d bytes", maxXBOWResponseBytes)
        }

        var findings []Finding
        if err := json.Unmarshal(responseBody, &findings); err != nil {
                return nil, fmt.Errorf("decode XBOW findings response: %w", err)
        }
        if findings == nil {
                return nil, fmt.Errorf("decode XBOW findings response: expected a JSON array")
        }

        file, err := os.OpenFile(x.LogPath, os.O_APPEND|os.O_CREATE|os.O_WRONLY, 0600)
        if err != nil {
                return findings, fmt.Errorf("open security event log: %w", err)
        }
        for _, finding := range findings {
                entry := xbowFindingLogEntry{
                        Timestamp:   time.Now().UTC().Format(time.RFC3339),
                        Source:      "XBOW",
                        EventID:     finding.ID,
                        AssetID:     assetID,
                        Severity:    finding.Severity,
                        Name:        finding.Name,
                        CVE:         finding.CVE,
                        CWE:         finding.CWE,
                        Description: finding.Description,
                }
                line, err := json.Marshal(entry)
                if err != nil {
                        _ = file.Close()
                        return findings, fmt.Errorf("encode XBOW finding for security log: %w", err)
                }
                line = append(line, '\n')
                written, err := file.Write(line)
                if err != nil {
                        _ = file.Close()
                        return findings, fmt.Errorf("write XBOW finding to security log: %w", err)
                }
                if written != len(line) {
                        _ = file.Close()
                        return findings, fmt.Errorf("write XBOW finding to security log: %w", io.ErrShortWrite)
                }
        }
        if err := file.Close(); err != nil {
                return findings, fmt.Errorf("close security event log: %w", err)
        }
        return findings, nil
}
